export default {
  async fetch(request, env, ctx) {
    // Hosting su GitHub Pages (dominio default) + localhost per lo sviluppo locale.
    // In produzione impostare comunque ALLOWED_ORIGINS con i domini reali.
    const allowedOrigin = env.ALLOWED_ORIGIN || 'https://riefolog-cyber.github.io';
    const allowedOrigins = (env.ALLOWED_ORIGINS || allowedOrigin + ',http://localhost:5173')
      .split(',')
      .map(s => s.trim())
      .filter(Boolean);

    const origin = request.headers.get('Origin');

    const makeCorsHeaders = (origin) => ({
      'Access-Control-Allow-Origin': origin || allowedOrigin,
      'Access-Control-Allow-Methods': 'POST, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type, Authorization',
      'Vary': 'Origin'
    });

    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: makeCorsHeaders(origin) });
    }

    if (request.method !== 'POST') {
      return new Response('Solo POST consentito', { status: 405, headers: makeCorsHeaders(origin) });
    }

    if (!origin || !allowedOrigins.includes(origin)) {
      return new Response(JSON.stringify({
        error: 'Origine non autorizzata',
        // Lista degli origin attualmente consentiti: utile per diagnosticare
        // subito i 403 senza dover aprire la console Cloudflare.
        allowedOrigins: allowedOrigins,
        hint: 'Aggiungi questa origine alla variabile ALLOWED_ORIGINS del Worker.'
      }), {
        status: 403, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
      });
    }

    const authResult = await verifyFirebaseToken(request, env);
    if (!authResult.ok) {
      console.error('[worker] Auth failed:', authResult.code, authResult.message);
      return new Response(JSON.stringify({ error: 'Autenticazione richiesta o non valida', code: authResult.code }), {
        status: 401, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
      });
    }
    const uid = authResult.uid;

    // Verifica server-side che l'utente sia un docente (prof).
    // L'accesso AI è riservato esclusivamente al ruolo prof.
    // IMPORTANTE: checkIsProf deve autenticarsi a Firestore col Firebase ID
    // Token del richiedente (Bearer): Firestore REST accetta i Firebase ID
    // Token e valuta le rules con request.auth corretto. Senza token Firestore
    // risponde PERMISSION_DENIED (le rules richiedono isAuth()) e il ruolo
    // risulterebbe sempre non-prof, anche per il docente legittimo.
    if (!(await checkIsProf(uid, request, env))) {
      return new Response(JSON.stringify({ error: 'Accesso AI riservato ai docenti.' }), {
        status: 403, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
      });
    }

    let groqDiagnostics = { status: null, error: null };
    let openrouterDiagnostics = { status: null, error: null };

    try {
      const body = await request.json();
      const { type, content, options = {} } = body;

      if (!content || typeof content !== 'string') {
        return new Response(JSON.stringify({ error: 'Campo "content" obbligatorio e deve essere testo.' }), {
          status: 400, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
        });
      }

      if (content.length > 60000) {
        return new Response(JSON.stringify({ error: 'Testo in ingresso troppo lungo (supera i 60000 caratteri).' }), {
          status: 400, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
        });
      }

      const now = Date.now();
      const limitKey = `${uid}:${type || 'default'}`;
      if (!globalThis._rateLimit) globalThis._rateLimit = new Map();
      const last = globalThis._rateLimit.get(limitKey) || 0;
      if (now - last < 10000) {
        return new Response(JSON.stringify({ error: 'Troppe richieste. Riprova tra qualche secondo.' }), {
          status: 429, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
        });
      }
      globalThis._rateLimit.set(limitKey, now);

      // Cleanup periodico per evitare crescita indefinita della mappa (rate limit)
      if (globalThis._rateLimit.size % 100 === 0) {
        const cutoff = now - 60000;
        for (const [key, ts] of globalThis._rateLimit) {
          if (ts < cutoff) globalThis._rateLimit.delete(key);
        }
      }

      const prompts = {
        quiz: `Sei un assistente didattico. Genera esattamente 3 domande a risposta multipla basandoti sul testo fornito. Restituisci RIGIDAMENTE un oggetto JSON con questa identica struttura: { "questions": [ { "question": "Testo della domanda...", "options": ["Opzione A","Opzione B","Opzione C","Opzione D"], "correctIndex": 0, "explanation": "Spiegazione della risposta..." } ] }. Livello: ${options.level || 'medio'}. Materia: ${options.subject || 'Generale'}.`,

        summary: `Sei un assistente didattico esperto. Riassumi il testo fornito dall'utente in esattamente 3 punti chiave essenziali, chiari e ben distanziati, ideali per studenti delle scuole superiori.
Regole di formattazione tassative (NON USARE ASSOLUTAMENTE TAG HTML O TABELLE):
- Inizia ogni punto chiave direttamente con una riga vuota di separazione dal precedente (doppio invio).
- Non usare asterischi per i grassetti, cancelletti per i titoli o barre verticali per le tabelle. Scrivi in testo semplice e pulito.
- Evita introduzioni e conclusioni.`,

        explanation: `Sei un assistente didattico. Spiega il concetto o il testo fornito dall'utente in modo semplice per studenti delle scuole superiori, con un esempio pratico della vita quotidiana.
Regole di formattazione tassative (NON USARE TAG HTML O TABELLE):
- Dividi la spiegazione in 2 o al massimo 3 brevi paragrafi separati da una riga vuota.
- Limita l'output a un massimo di 150 parole.
- Non usare liste puntate, elenchi, asterischi o blocchi di codice. Solo testo narrativo fluido ed elegante.`,

        json: `Sei un assistente didattico specializzato nella generazione di risposte JSON strutturate.
Devi analizzare l'input dell'utente ed eseguire ESCLUSIVAMENTE la specifica richiesta descritta nel suo messaggio.
ATTENZIONE MASSIMA: Se l'utente ti chiede di analizzare i commenti, i dubbi o la discussione degli studenti, devi dare priorità assoluta e totale ai commenti reali forniti dagli studenti. Basa l'analisi unicamente sulle reali esigenze, dubbi, o risposte emersi dai commenti, ignorando la lezione generica principale se non strettamente pertinente.
Restituisci il risultato ESCLUSIVAMENTE sotto forma di oggetto JSON valido e strutturato, seguendo la struttura richiesta dall'utente, senza aggiungere alcuna spiegazione testuale prima o dopo, e senza usare blocchi di codice.`,

        text: `Sei un assistente didattico esperto, preciso e disponibile. Rispondi alle richieste dell'utente in modo strutturato.
ATTENZIONE RIGIDISSIMA:
Se il messaggio dell'utente contiene alla fine la sezione "Chi ha commentato:" e un elenco di "COMMENTI" e ti viene chiesta l'analisi della partecipazione (es. "Chi ha partecipato di più?", "Quali concetti emergono maggiormente?", "Ci sono errori concettuali?"), devi IGNORARE le statistiche contenute nella lezione. Devi analizzare ESCLUSIVAMENTE l'elenco dei nomi e dei commenti degli studenti reali che trovi in fondo al messaggio per formulare la tua risposta.
Regole di formattazione tassative (NON USARE TAG HTML O TABELLE):
- Organizza le spiegazioni in paragrafi brevi e separali sempre con una riga vuota (doppio invio).
- Usa esclusivamente trattini semplici per gli elenchi. Non usare asterischi per evidenziare i testi.
- Rendi la risposta visivamente leggera, pulita e immediata. Non inserire blocchi di codice con tre backtick.`
      };

      const systemPrompt = prompts[type] || prompts.summary;
      const isStructured = (type === 'quiz' || type === 'json');
      const temperature = isStructured ? 0.2 : 0.6;
      const maxTokens = options.max_tokens || (type === 'quiz' ? 1200 : (type === 'json' ? 1500 : 2000));

      let aiSuccess = false;
      let aiText = null;
      let aiProvider = null;
      let aiModel = null;

      // 1. Prova prima GROQ, iterando i modelli candidati in ordine di priorità.
      //    Se un modello fallisce (HTTP 4xx/5xx o risposta vuota), si passa al
      //    successivo e lo si declassa in cache.
      const responseFormatParam = isStructured ? { type: 'json_object' } : undefined;
      const callParams = {
        messages: [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: content }
        ],
        temperature: temperature,
        maxTokens: maxTokens,
        responseFormatParam: responseFormatParam
      };

      if (env.GROQ_API_KEY) {
        const models = await getModelCandidates('groq', env, 'llama-3.3-70b-versatile');
        for (const model of models) {
          const r = await callProvider('groq', env, allowedOrigin, model, callParams);
          if (r.ok) {
            aiText = r.text;
            aiSuccess = true;
            aiProvider = 'groq';
            aiModel = model;
            break;
          }
          groqDiagnostics = { status: r.status, error: r.error, model: model };
          demoteModel('groq', model);
          console.error('[worker] Groq modello fallito:', model, r.status, r.error);
        }
      }

      // 2. OpenRouter Fallback (stessa logica di iterazione sui candidati)
      // Solo modelli free: getModelCatalog filtra già su pricing 0/0, fallback
      // esplicito è il primo della priority free (non 'openrouter/free' generico).
      if (!aiSuccess && env.OPENROUTER_API_KEY) {
        const models = await getModelCandidates('openrouter', env, OPENROUTER_PRIORITY[0]);
        for (const model of models) {
          const r = await callProvider('openrouter', env, allowedOrigin, model, callParams);
          if (r.ok) {
            aiText = r.text;
            aiSuccess = true;
            aiProvider = 'openrouter';
            aiModel = model;
            break;
          }
          openrouterDiagnostics = { status: r.status, error: r.error, model: model };
          demoteModel('openrouter', model);
          console.error('[worker] OpenRouter modello fallito:', model, r.status, r.error);
        }
      }

      // Controllo finale di disponibilità (con diagnostica utile per il debug)
      if (!aiSuccess || !aiText) {
        return new Response(JSON.stringify({
          error: 'Servizio di AI momentaneamente non disponibile, riprova più tardi.',
          diagnostics: { groq: groqDiagnostics, openrouter: openrouterDiagnostics }
        }), {
          status: 503, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
        });
      }

      // Rimuovi eventuali delimitatori markdown dalla risposta strutturata
      if (isStructured) {
        aiText = aiText.trim();
        if (aiText.startsWith('```json')) aiText = aiText.replace(/^```json/, '');
        else if (aiText.startsWith('```')) aiText = aiText.replace(/^```/, '');
        if (aiText.endsWith('```')) aiText = aiText.replace(/```$/, '');
        aiText = aiText.trim();
      }

      // Risposta successo: diagnostica visibile solo con ?debug=1 (query Worker o ?debug=1 nel frontend)
      // così il Network del browser mostra direttamente quale provider/modello ha vinto.
      const debugOn = new URL(request.url).searchParams.get('debug') === '1';
      const successBody = { success: true, data: aiText };
      if (debugOn) {
        successBody.diagnostics = {
          provider: aiProvider,
          model: aiModel,
          groq: groqDiagnostics,
          openrouter: openrouterDiagnostics
        };
      }
      return new Response(JSON.stringify(successBody), {
        status: 200, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
      });

    } catch (err) {
      return new Response(JSON.stringify({ error: 'Operazione non riuscita o errore generico' }), {
        status: 500, headers: { ...makeCorsHeaders(origin), 'Content-Type': 'application/json' }
      });
    }
  }
};

/**
 * ── SELEZIONE DINAMICA DEL MODELLO AI ───────────────────────────────────────
 * Il worker sceglie AUTOMATICAMENTE il miglior modello gratuito disponibile
 * interrogando i cataloghi di Groq/OpenRouter, con cache di 24h.
 *
 * Come funziona:
 *  - Se imposti GROQ_MODEL / OPENROUTER_MODEL nelle env vars, quello vince
 *    (override esplicito, nessuna chiamata al catalogo).
 *  - Altrimenti il worker scarica il catalogo modelli e costruisce una lista
 *    ORDINATA di candidati (i modelli prioritari presenti nel catalogo, poi un
 *    eventuale modello "noto"). La richiesta prova i candidati in sequenza:
 *    se il primo fallisce (HTTP 4xx/5xx o risposta vuota), passa al successivo
 *    e lo "declassa" in cache, così il fallimento non viene ripetuto.
 *  - Se il catalogo non è raggiungibile si ripiega sul fallback passato.
 */
const MODEL_CACHE_TTL_MS = 24 * 60 * 60 * 1000; // 24h: i cataloghi cambiano raramente
const MODEL_FAIL_TTL_MS = 10 * 60 * 1000; // cache negativa: 10min dopo un errore catalogo

// Modelli dismessi/sospesi dal provider: vanno esclusi anche dal fallback "noto".
// Es. Groq ha decommissionato llama-3.1-8b-instant il 16/08/2026.
const MODEL_DENYLIST = new Set([
  'llama-3.1-8b-instant',
]);

// Modelli Groq gratuiti ordinati per capacità (il primo disponibile vince).
// Riordina questa lista se la qualità percepita di un modello cambia.
const GROQ_PRIORITY = [
  'openai/gpt-oss-120b',
  'llama-4-scout',
  'llama-3.3-70b-versatile',
  'openai/gpt-oss-20b',
];

// Modelli OpenRouter gratuiti (:free) ordinati per capacità.
// Aggiornati al catalogo live 2026-08: solo modelli con pricing 0/0 (verificati
// via https://openrouter.ai/api/v1/models). I precedenti deepseek/llama/qwen
// non sono più free → sostituiti. L'ordine privilegia modelli chat istruiti
// e grandi (550B/120B) poi generalisti. Esclusi lyria (TTS) e openrouter/free
// generico (non deterministico).
const OPENROUTER_PRIORITY = [
  'z-ai/glm-5.2:free',
  'nvidia/nemotron-3-ultra-550b-a55b:free',
  'nvidia/nemotron-3-super-120b-a12b:free',
  'minimax/minimax-m3:free',
  'minimax/minimax-m2.7:free',
  'google/gemma-4-26b-a4b-it:free',
  'google/gemma-4-31b-it:free',
  'nvidia/nemotron-3.5-lightning:free',
  'nvidia/nemotron-3-nano-omni-30b-a3b-reasoning:free',
  'cohere/north-mini-code:free',
  'liquid/lfm-2.5-2.6b:free',
  'dots-studio/dots-3-note-preview:free',
  'poolside/laguna-s-2.1:free',
  'poolside/laguna-xs-2.1:free',
  'thinkingmachines/inkling:free',
  'thinkingmachines/inkling-small:free',
  'nvidia/nemotron-3.5-content-safety:free',
];

// Fallback per modelli chat: qualsiasi modello che termina con :free.
// Sostituisce la precedente whitelist per prefisso (llama/qwen/...) che non
// copriva i provider attuali (nvidia, z-ai, cohere, liquid...). Garantisce
// che il fallback resti sempre su un modello gratuito anche se la priority
// diventa obsoleta in futuro.
const KNOWN_MODEL_PREFIX = /:free$/;

// Scarica l'elenco dei modelli dal provider, escludendo i modelli dismessi.
// Per OpenRouter filtra SOLO i modelli gratuiti (pricing 0/0) per non finire
// mai su un modello a pagamento.
async function getModelCatalog(provider, env) {
  let ids;
  if (provider === 'groq') {
    const res = await fetch('https://api.groq.com/openai/v1/models', {
      headers: { 'Authorization': `Bearer ${env.GROQ_API_KEY}` }
    });
    if (!res.ok) throw new Error('catalogo Groq HTTP ' + res.status);
    const data = await res.json();
    ids = (data.data || []).map((m) => m.id);
  } else {
    const res = await fetch('https://openrouter.ai/api/v1/models');
    if (!res.ok) throw new Error('catalogo OpenRouter HTTP ' + res.status);
    const data = await res.json();
    ids = (data.data || [])
      .filter((m) => {
        const p = m.pricing || {};
        return p.prompt === '0' && p.completion === '0';
      })
      .map((m) => m.id);
  }
  return ids.filter((id) => !MODEL_DENYLIST.has(id));
}

// Costruisce la lista ordinata dei modelli candidati per un provider.
// Il primo è il modello preferito (override esplicito → cache → catalogo →
// fallback), i successivi sono le alternative da provare in caso di fallimento.
async function getModelCandidates(provider, env, fallback) {
  // Override esplicito via env var: nessuna chiamata al catalogo
  const explicit = provider === 'groq' ? env.GROQ_MODEL : env.OPENROUTER_MODEL;
  if (explicit) return [explicit];

  const cache = globalThis._modelCache;
  // Gli errori usano una cache negativa breve: un'eventuale indisponibilità
  // del catalogo NON deve far riprovare la fetch a ogni singola richiesta.
  const ttl = cache && cache[provider] && cache[provider].failed ? MODEL_FAIL_TTL_MS : MODEL_CACHE_TTL_MS;
  if (cache && cache[provider] && Array.isArray(cache[provider].candidates) &&
      Date.now() - cache[provider].ts < ttl) {
    return cache[provider].candidates;
  }

  try {
    const catalog = await getModelCatalog(provider, env);
    const priority = provider === 'groq' ? GROQ_PRIORITY : OPENROUTER_PRIORITY;
    // 1) modelli prioritari presenti nel catalogo, in ordine di lista
    const ranked = priority.filter((id) => catalog.includes(id));
    // 2) fallback su un modello chat "noto" (solo se non già in lista)
    const known = catalog.find((id) => KNOWN_MODEL_PREFIX.test(id));
    const candidates = known && !ranked.includes(known) ? [...ranked, known] : ranked;
    if (candidates.length) {
      globalThis._modelCache = globalThis._modelCache || {};
      globalThis._modelCache[provider] = { candidates: candidates, ts: Date.now(), failed: false };
      console.log('[worker] Modelli dinamici ' + provider + ': ' + candidates.join(', '));
      return candidates;
    }
  } catch (err) {
    console.error('[worker] Catalogo ' + provider + ' non disponibile:', err.message);
  }
  globalThis._modelCache = globalThis._modelCache || {};
  globalThis._modelCache[provider] = { candidates: [fallback], ts: Date.now(), failed: true };
  return [fallback];
}

// Declassa un modello fallito: lo rimuove dalla lista candidati in cache così
// la prossima richiesta non lo riprova subito. Se non resta alcun candidato,
// marca la cache come negativa (breve TTL) per ricostruire la lista dal catalogo.
function demoteModel(provider, failedModel) {
  const cache = globalThis._modelCache && globalThis._modelCache[provider];
  if (!cache || !Array.isArray(cache.candidates)) return;
  const next = cache.candidates.filter((m) => m !== failedModel);
  globalThis._modelCache[provider] = next.length
    ? { candidates: next, ts: Date.now(), failed: false }
    : { candidates: [failedModel], ts: Date.now(), failed: true };
}

// Esegue UNA chiamata al provider per un dato modello. Restituisce { ok, text }
// oppure { ok: false, status, error } senza mai lanciare. Gestisce il retry sul
// response_format 400 (valido per Groq E OpenRouter).
async function callProvider(provider, env, allowedOrigin, model, params) {
  const isGroq = provider === 'groq';
  const url = isGroq
    ? 'https://api.groq.com/openai/v1/chat/completions'
    : 'https://openrouter.ai/api/v1/chat/completions';
  const headers = isGroq
    ? { 'Authorization': `Bearer ${env.GROQ_API_KEY}`, 'Content-Type': 'application/json' }
    : {
        'Authorization': `Bearer ${env.OPENROUTER_API_KEY}`,
        'Content-Type': 'application/json',
        'HTTP-Referer': allowedOrigin,
        'X-Title': 'ScuolaBoard'
      };

  const buildBody = (withFormat) => JSON.stringify({
    model: model,
    messages: params.messages,
    temperature: params.temperature,
    ...(withFormat && params.responseFormatParam ? { response_format: params.responseFormatParam } : {}),
    max_tokens: params.maxTokens
  });

  try {
    let res = await fetch(url, { method: 'POST', headers: headers, body: buildBody(true) });

    // Retry logico in caso di 400 da response_format (modello senza json_object)
    if (!res.ok && params.responseFormatParam && res.status === 400) {
      res = await fetch(url, { method: 'POST', headers: headers, body: buildBody(false) });
    }

    if (res.ok) {
      const data = await res.json();
      const text = data.choices && data.choices[0] && data.choices[0].message && data.choices[0].message.content;
      if (text) return { ok: true, text: text };
      return { ok: false, status: res.status, error: 'Risposta vuota dal modello ' + model };
    }
    return { ok: false, status: res.status, error: 'HTTP ' + res.status };
  } catch (err) {
    return { ok: false, status: null, error: err.message || String(err) };
  }
}

/**
 * Verifica il Firebase ID Token JWT in un Cloudflare Worker.
 * ORA verifica la firma crittografica RS256 contro le chiavi pubbliche Google
 * (JWKS da https://www.googleapis.com/oauth2/v3/certs) PRIMA di accettare
 * qualsiasi claim. Cache JWKS con TTL 1 ora + skew tolerance 60s sull'exp.
 *
 * Sicurezza: chiunque può forgiare un JWT con i claim corretti, ma solo uno
 * firmato da Google (con la chiave privata del progetto Firebase) supera
 * la verifica. Questo elimina l'impersonazione arbitraria di UID.
 */
async function verifyFirebaseToken(request, env) {
  const authHeader = request.headers.get('Authorization');
  if (!authHeader) {
    return { ok: false, code: 'AUTH_NO_HEADER', message: 'Header Authorization mancante' };
  }
  if (!authHeader.startsWith('Bearer ')) {
    return { ok: false, code: 'AUTH_BAD_FORMAT', message: 'Formato header authorization errato' };
  }

  const token = authHeader.substring(7);
  const parts = token.split('.');
  if (parts.length !== 3) {
    return { ok: false, code: 'AUTH_JWT_MALFORMED', message: 'Token JWT non valido' };
  }

  // Util: base64url → bytes / stringa
  const b64UrlToBytes = (s) => {
    let b64 = s.replace(/-/g, '+').replace(/_/g, '/');
    while (b64.length % 4) b64 += '=';
    const bin = atob(b64);
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return bytes;
  };
  const b64UrlToStr = (s) => {
    let b64 = s.replace(/-/g, '+').replace(/_/g, '/');
    while (b64.length % 4) b64 += '=';
    return atob(b64);
  };

  // 1. Ottieni/aggiorna cache JWKS (TTL 1h)
  // IMPORTANTE: per i Firebase ID Token (iss: securetoken.google.com/<project>)
  // le chiavi pubbliche NON stanno su oauth2/v3/certs (che serve i Google Account
  // ID Tokens di "Sign in with Google"), bensi sull'endpoint Secure Token Service:
  //   https://www.googleapis.com/service_accounts/v1/jwk/securetoken@system.gserviceaccount.com
  // (Firebase Admin SDK usa la variante x509 allo stesso URL: robot/v1/metadata/x509/...)
  const FIREBASE_JWKS_URL = 'https://www.googleapis.com/service_accounts/v1/jwk/securetoken@system.gserviceaccount.com';
  const nowMs = Date.now();
  if (!globalThis._jwksCache || (nowMs - globalThis._jwksCache.ts) > 3600000) {
    try {
      const jwksRes = await fetch(FIREBASE_JWKS_URL);
      if (!jwksRes.ok) {
        return { ok: false, code: 'JWKS_FETCH_FAILED', message: 'Impossibile recuperare chiavi pubbliche Google (HTTP ' + jwksRes.status + ')' };
      }
      const jwks = await jwksRes.json();
      globalThis._jwksCache = { keys: jwks.keys || [], ts: nowMs };
    } catch (e) {
      return { ok: false, code: 'JWKS_FETCH_FAILED', message: 'Errore fetch JWKS: ' + (e.message || String(e)) };
    }
  }
  const jwksKeys = globalThis._jwksCache.keys;

  // 2. Decodifica header e forza RS256 (rifiuta altri alg → sicurezza)
  let header;
  try {
    header = JSON.parse(b64UrlToStr(parts[0]));
  } catch (e) {
    return { ok: false, code: 'AUTH_DECODE_ERROR', message: 'Header JWT non decodificabile' };
  }
  if (header.alg !== 'RS256') {
    return { ok: false, code: 'AUTH_BAD_ALG', message: 'Algoritmo non supportato (solo RS256 accettato)' };
  }

  const jwk = jwksKeys.find((k) => k.kid === header.kid);
  if (!jwk) {
    return { ok: false, code: 'JWKS_NO_KID_MATCH', message: 'Nessuna chiave pubblica corrispondente al kid del token' };
  }

  // 3. Verifica firma RS256 via Web Crypto API
  let signatureValid = false;
  try {
    const pubKey = await crypto.subtle.importKey(
      'jwk', jwk,
      { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' },
      false, ['verify']
    );
    const signature = b64UrlToBytes(parts[2]);
    const signedData = new TextEncoder().encode(parts[0] + '.' + parts[1]);
    signatureValid = await crypto.subtle.verify(
      'RSASSA-PKCS1-v1_5', pubKey, signature, signedData
    );
  } catch (e) {
    return { ok: false, code: 'JWKS_BAD_SIGNATURE', message: 'Errore verifica firma: ' + (e.message || String(e)) };
  }
  if (!signatureValid) {
    return { ok: false, code: 'JWKS_BAD_SIGNATURE', message: 'Firma JWT non valida (rifiutato)' };
  }

  // 4. Payload + claim check (SOLO DOPO firma valida)
  let payload;
  try {
    payload = JSON.parse(b64UrlToStr(parts[1]));
  } catch (e) {
    return { ok: false, code: 'AUTH_DECODE_ERROR', message: 'Payload JWT non decodificabile' };
  }

  // Skew 60s tollerato per clock drift server/client
  const nowSec = Math.floor(nowMs / 1000);
  if (!payload.exp || nowSec > (payload.exp + 60)) {
    return { ok: false, code: 'AUTH_TOKEN_EXPIRED', message: 'Token scaduto' };
  }
  // iat opzionale: se presente, non deve essere nel futuro (skew 60s)
  if (payload.iat && (payload.iat - 60) > nowSec) {
    return { ok: false, code: 'AUTH_BAD_IAT', message: 'iat nel futuro' };
  }

  // projectId ancora da env (rimossa la nota "fallback hardcoded" perché la
  // verifica firma copre già l'autenticità del progetto).
  const projectId = env.FIREBASE_PROJECT_ID || 'scuolaboard-874d4';
  if (payload.aud !== projectId) {
    return { ok: false, code: 'AUTH_BAD_AUDIENCE', message: 'Audience non valido' };
  }
  if (payload.iss !== `https://securetoken.google.com/${projectId}`) {
    return { ok: false, code: 'AUTH_BAD_ISSUER', message: 'Issuer non valido' };
  }
  if (!payload.sub || typeof payload.sub !== 'string' || payload.sub.length > 128) {
    return { ok: false, code: 'AUTH_NO_SUB', message: 'Manca identificativo sub/uid (o troppo lungo)' };
  }

  return { ok: true, uid: payload.sub };
}

/**
 * Verifica se l'utente con il dato uid ha ruolo "prof" in Firestore.
 * Utilizzato per il gating server-side dell'accesso AI.
 *
 * Passa il Firebase ID Token del richiedente come Bearer: senza di esso
 * Firestore REST risponde 403 PERMISSION_DENIED (le rules richiedono
 * isAuth()) e checkIsProf fallirebbe sempre. Con il token, la rule
 * `allow read: if isAuth() && (request.auth.uid == uid || isProf())`
 * consente all'utente di leggere il PROPRIO profilo e leggere il ruolo.
 */
async function checkIsProf(uid, request, env) {
  const projectId = env.FIREBASE_PROJECT_ID || 'scuolaboard-874d4';
  const url = `https://firestore.googleapis.com/v1/projects/${projectId}/databases/(default)/documents/users/${uid}`;
  const authHeader = request.headers.get('Authorization');
  try {
    const res = await fetch(url, {
      method: 'GET',
      headers: {
        'Content-Type': 'application/json',
        // Firebase ID Token del richiedente (già verificato prima da verifyFirebaseToken)
        ...(authHeader ? { 'Authorization': authHeader } : {})
      }
    });
    if (!res.ok) {
      console.error('[worker] checkIsProf Firestore HTTP', res.status, (await res.text()).slice(0, 200));
      return false;
    }
    const doc = await res.json();
    return doc.fields?.role?.stringValue === 'prof';
  } catch (e) {
    console.error('[worker] checkIsProf error:', e.message);
    return false;
  }
}
